cmake_minimum_required(VERSION 3.20)
project(Shopping_Portal VERSION 2.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(commerce INTERFACE)
target_include_directories(commerce INTERFACE include)
add_executable(catalog_cli src/catalog_cli.cpp)
target_link_libraries(catalog_cli PRIVATE commerce)
add_executable(domain_tests tests/domain_tests.cpp)
target_link_libraries(domain_tests PRIVATE commerce)
add_executable(analytics_tests tests/analytics_tests.cpp)
target_link_libraries(analytics_tests PRIVATE commerce)
enable_testing()
add_test(NAME domain_correctness COMMAND domain_tests)
add_test(NAME catalog_seed COMMAND catalog_cli --data ${CMAKE_SOURCE_DIR}/data/sample)
add_test(NAME analytics_correctness COMMAND analytics_tests ${CMAKE_SOURCE_DIR}/data/sample)
add_test(NAME cli_quote COMMAND catalog_cli --data ${CMAKE_SOURCE_DIR}/data/sample --quote Apples:2)
set_tests_properties(cli_quote PROPERTIES PASS_REGULAR_EXPRESSION "TOTAL PKR +330\\.00")
add_test(NAME cli_stats COMMAND catalog_cli --data ${CMAKE_SOURCE_DIR}/data/sample --stats)
set_tests_properties(cli_stats PROPERTIES PASS_REGULAR_EXPRESSION "All categories +351")
add_test(NAME cli_rejects_overselling COMMAND catalog_cli --data ${CMAKE_SOURCE_DIR}/data/sample --quote Apples:999)
set_tests_properties(cli_rejects_overselling PROPERTIES WILL_FAIL TRUE)
if(WIN32)
  add_executable(Shopping_Portal src/Main.cpp src/Admin.cpp src/Vendor.cpp src/Customer.cpp src/Items.cpp src/Cart.cpp)
  target_compile_definitions(Shopping_Portal PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
  target_link_libraries(Shopping_Portal PRIVATE commerce user32)
  add_executable(console_tests tests/unit_console.cpp src/Items.cpp src/Cart.cpp)
  target_compile_definitions(console_tests PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
  target_link_libraries(console_tests PRIVATE commerce user32)
  file(MAKE_DIRECTORY ${CMAKE_BINARY_DIR}/unit_fixture)
  file(COPY data/sample/ DESTINATION ${CMAKE_BINARY_DIR}/unit_fixture)
  add_test(NAME console_correctness COMMAND console_tests)
  set_tests_properties(console_correctness PROPERTIES WORKING_DIRECTORY ${CMAKE_BINARY_DIR}/unit_fixture)
endif()
